numeric_cols = ("bytes_in", "bytes_out", "dst_port")
display(Markdown("STEP: Convert numeric columns safely"))

for ncol in numeric_cols:
    if ncol in df.columns:
        # Report before state
        before_dtype = df[ncol].dtype
        before_na = df[ncol].isna().sum()
        non_numeric_count = df[ncol].apply(lambda x: not (pd.isna(x) or isinstance(x, (int, float, np.integer, np.floating)))).sum()
        
        # Convert safely: coerce -> fillna(0) -> convert to int
        df[ncol] = pd.to_numeric(df[ncol], errors="coerce").fillna(0).astype(int)
        
        # Report after state
        after_dtype = df[ncol].dtype
        after_na = df[ncol].isna().sum()
        zero_count = (df[ncol] == 0).sum()
        
        print(f"\nColumn: {ncol}")
        print(f" - dtype before: {before_dtype}, non-null before: {len(df) - before_na}, non-numeric entries: {non_numeric_count}")
        print(f" - dtype after:  {after_dtype}, nulls after: {after_na}, zeros after: {zero_count} (may include coerced/missing values)")
    else:
        print(f"{ncol}: not found - skipped")

for tcol in ("creation_time", "end_time"):
    if tcol in df.columns and not pd.api.types.is_datetime64_any_dtype(df[tcol]):
        df[tcol] = pd.to_datetime(df[tcol], errors="coerce")

# Compute session_duration when both columns exist
if ("creation_time" in df.columns) and ("end_time" in df.columns):
    df["session_duration"] = (df["end_time"] - df["creation_time"]).dt.total_seconds()
    print(f"Computed session_duration for rows with both timestamps: {df['session_duration'].notna().sum()}")
else:
    df["session_duration"] = np.nan

neg_count = (df["session_duration"] <= 0).sum()
df.loc[df["session_duration"] <= 0, "session_duration"] = np.nan
print(f"Replaced {neg_count} non-positive durations with NaN")

# Fill NaNs with median if available, otherwise with 600s default
if df["session_duration"].notna().any():
    med = df["session_duration"].median()
else:
    med = 600
df["session_duration"] = df["session_duration"].fillna(med)

# Safe conversion helper
def safe_int_col(df, col):
    if col in df.columns:
        before_non_null = df[col].notna().sum()
        df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0).astype(int)
        after_non_null = df[col].notna().sum()
        return before_non_null, after_non_null
    else:
        df[col] = 0
        return 0, len(df)
